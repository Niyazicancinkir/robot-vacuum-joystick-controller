      import os

      DEVICE_ID = "1069182766"
      SIID_DIRECTION = 7
      PIID_DIRECTION = 16

      def send_direction(vacuum, direction):
        vacuum.send("set_properties", [{"did": DEVICE_ID,
          "siid": SIID_DIRECTION, "piid": PIID_DIRECTION, "value": direction}])

      # 1=ileri, 2=sol, 3=sağ, 4=geri, 5=dur, 10=çıkış
      send_direction(vacuum, 1)
